*cgget

  csp -1
  getparam $11,$12,$13,$14,$15,$21,$22,$23,$24,$25,$31,$32,$33,$34,$35,$41,$42,$43,$44,$45

  ;%8 Y坐标 %9 X坐标
  for %8 = 0 to 3
    for %9 = 0 to 4
      mov %7,10+10*%8+%9+1
      mov %10,%7+100
        mov %%10,0 ;复位
        if fchk "background/ev_"+$%7+".jpg" mov %%10,1 ;检查是否经历CG图片
    next
  next

return

*cgget_btn_loop

  exbtn_d "P710-749"

  ;%8 Y坐标 %9 X坐标

      mov %10,%7+100
      itoa $7,%7

    next

  exbtn 898,59

  if %0 = 59 && %6<7 mov %6,%6+1 : itoa $6,%6 : goto "*cg_"+$6
  if %0 = 58 && %6>1 mov %6,%6-1 : itoa $6,%6 : goto "*cg_"+$6

*cgview2
  getparam $0,%0,%1,%2,%3
  btndef "background\"+$0+".jpg"
  mov %4,0
  mov %5,0
*showloop

  wait 50
  add %4,%2
  add %5,%3
if %4<=%0-800 && %5<=%1-600 goto *showloop
  click
  sub %4,%2
  sub %5,%3
*showloop2
  blt 0,0,800,600,%4,%5,800,600
  wait 50
  sub %4,%2
  sub %5,%3
if %4>=0 && %5>=0 goto *showloop2
click

lsp 900,":c;esystem/album_bg1.bmp",0,0,255
print 1
