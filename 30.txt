*get_image
	getparam s%601
	startswith %602, $%601, ":"
	if %602 > 0 return
	if $%601 = "sword_01" goto *get_image_a
	if $%601 = "re_cloud" goto *get_image_a
	if $%601 = "particle" goto *get_image_a
	startswith %602, $%601, "aiy"
	startswith %603, $%601, "chs"
	add %602, %603
	if %602 <= 0 goto *get_image_try
	right $601, $%601, -3
	if $601 = "10070_01" goto *get_image_c
	if $601 = "10090_01" goto *get_image_c
	if $601 = "10160_01" goto *get_image_c
	if $601 = "10190_10" goto *get_image_c
	if $601 = "10190_27" goto *get_image_c
	if $601 = "20200_13" goto *get_image_c
	if $601 = "20200_anm01" goto *get_image_a
	if $601 = "30100_01" goto *get_image_c
	if $601 = "30180_06" goto *get_image_c
	if $601 = "40360_01" goto *get_image_c
	if $601 = "50300_01" goto *get_image_c
	if $601 = "50310_01" goto *get_image_c
	if $601 = "50310_02" goto *get_image_c
	split $601, "_", $603, $602
	if $603 = "10000" goto *get_image_c
	if $602 = "sprite" goto *get_image_a
	if $602 = "roll" goto *get_image_a
	mid $602, $602, 0, 1
	atoi %602, $602
	if $602 != "0" & %602 = 0 & $602 != "t" goto *get_image_try
	goto *get_image_a
